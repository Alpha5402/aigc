### 4. 等待 plusready

App 端不要假设 `plus` 一定立刻存在。

请实现一个简单的 `waitForPlusReady()`：

```ts
const waitForPlusReady = () => {
  return new Promise<any>((resolve, reject) => {
    const currentPlus = (globalThis as any).plus

    if (currentPlus) {
      resolve(currentPlus)
      return
    }

    let done = false

    const timer = setTimeout(() => {
      if (done) return
      done = true
      reject(new Error('App 运行环境未准备完成，请稍后重试'))
    }, 3000)

    document.addEventListener(
      'plusready',
      () => {
        if (done) return
        done = true
        clearTimeout(timer)
        resolve((globalThis as any).plus)
      },
      false,
    )
  })
}
```

在 `startPlusSpeechRecognize()` 中先：

```ts
const plusObj = await waitForPlusReady()
```

### 5. 识别超时

H5 和 App 端都要加超时兜底。

建议 10 到 12 秒：

```ts
const timer = setTimeout(() => {
  // H5 调 recognition.stop()
  // App 调 plus.speech.stopRecognize()
  reject(new Error('长时间没有检测到语音，请重新尝试或使用文字输入'))
}, 12000)
```

注意避免重复 resolve / reject。

可以使用：

```ts
let settled = false

const finish = (callback: () => void) => {
  if (settled) return
  settled = true
  clearTimeout(timer)
  callback()
}
```

## 五、参考实现逻辑

你可以按下面思路实现，但要结合项目实际 TypeScript 配置微调。

```ts
type SpeechErrorMap = Record<string, string>

const speechErrorMessages: SpeechErrorMap = {
  'no-speech': '没有听到有效语音，请靠近麦克风后再试',
  'audio-capture': '没有检测到可用麦克风，请检查设备或关闭占用麦克风的软件',
  'not-allowed': '麦克风权限未开启，请在系统或浏览器中允许麦克风权限',
  'service-not-allowed': '当前环境不允许使用语音识别服务，请检查权限或改用文字输入',
  network: '语音识别服务连接失败，请检查网络后重试',
  aborted: '语音识别已中断，请重新点击语音按钮',
  'language-not-supported': '当前环境不支持中文语音识别，请使用文字输入',
}

const getSpeechErrorMessage = (error?: string, fallback?: string) => {
  if (error && speechErrorMessages[error]) return speechErrorMessages[error]
  return fallback || `语音识别失败${error ? `：${error}` : ''}，请使用文字输入`
}
```

H5 识别成功后返回文本。

App 识别成功后返回文本。

任何失败都抛出 Error，让组件捕获并显示文字输入兜底。

## 六、修改语音助手组件

找到当前语音助手组件中直接调用 `SpeechRecognition` 的地方，改成：

```ts
import { startVoiceRecognize } from '@/utils/voice-recognizer'
```

或者根据当前项目别名配置使用相对路径。

把原来的语音识别逻辑替换成：

```ts
const handleVoiceInput = async () => {
  if (listening.value) return

  listening.value = true
  assistantReply.value = '正在聆听，请说出你的需求...'
  showTextInput.value = false

  try {
    const text = await startVoiceRecognize()

    recognizedText.value = text
    assistantReply.value = `已识别：${text}`

    await sendAssistantMessage(text)
  } catch (error: any) {
    console.warn('[voice] recognize failed:', error)
    assistantReply.value = error?.message || '语音识别失败，请使用文字输入'
    showTextInput.value = true
  } finally {
    listening.value = false
  }
}
```

注意：

* `sendAssistantMessage(text)` 要替换成当前组件里原本发送到后端助手的方法
* 不要破坏现有 `/api/assistant/chat` 调用
* 如果当前组件变量名不是 `assistantReply / showTextInput / recognizedText`，按实际变量名适配
